}

async function closeOwnedLoopback(server: http.Server, sockets: Set<net.Socket>) {
  // Subscribe before destroy: server.close's callback can precede socket close events.
  const socketClosures = [...sockets].map(socket => new Promise<void>(yes => socket.once("close", () => yes())));
  let cleanupTimer: ReturnType<typeof setTimeout> | undefined;
  try {
    server.closeIdleConnections();
    const closed = new Promise<void>(yes => server.close(() => yes()));
    for (const socket of sockets) socket.destroy();
    await Promise.race([
      Promise.all([closed, ...socketClosures]),
      new Promise<never>((_, no) => { cleanupTimer = setTimeout(() => no(new Error("owned socket cleanup timeout")), SOCKET_CLEANUP_BUDGET_MS); }),
    ]);
  } finally { clearTimeout(cleanupTimer); }
}

async function main() {
  const before = { A: tree(roots.A), B: tree(roots.B) }; validateTrees(before.A, before.B);
  const loaded = { A: await modules(roots.A), B: await modules(roots.B) };
  const ready = { A: rehydrate(loaded.A).receipt, B: rehydrate(loaded.B).receipt }; assert.deepEqual(ready.A, ready.B);
  assert.equal(escapeAttempts, 0, "offline preflight attempted I/O");
  const declaration = { schemaVersion: 1, mode: "offline recorded-response validation; not latency/live/finalsim",
    runnerSha256: hash(fileURLToPath(import.meta.url)), artifacts, runtimes: before, source, search: spec.search, ready,
    recordings: { acceptedKeys: recordings.size, readRows, diagnosticRows, ignoredRows, conflicts: 0 },
    order: ["A", "B", "B", "A"], operationBudgetMs: 30_000, socketCleanupBudgetMs: SOCKET_CLEANUP_BUDGET_MS, maxWireItemsPerInvocation: 4096,
    exactTransport: { maxBatchSize: 64, maxConcurrentBatches: 8, schedulerCapacity: 12, producerReserved: 4 } };
  if (!options["--run"]) {
    console.log(json({ status: "check-passed", runnerSha256: declaration.runnerSha256, artifacts, source, ready,
      recordings: declaration.recordings, runtimes: Object.fromEntries(Object.entries(before).map(([arm, t]) =>
        [arm, { root: t.root, sha256: t.sha256, centralSha256: t.files[changed], unimportedTestSha256: t.files[unimportedTest] }])) }));
    return;
  }
  const out = options["--out"] ?? path.join(E, "source-storage-offline-2");
  assert.equal(path.dirname(out), E); fs.mkdirSync(out);
  fs.writeFileSync(path.join(out, "declaration.json"), json(declaration), { flag: "wx" });
  const results: any[] = []; let error: string | null = null;
  try {
    for (const arm of declaration.order as ("A" | "B")[]) {
      if (results.some(r => r.status !== "complete")) { results.push({ arm, status: "not-run-after-failure" }); continue; }
      results.push(await invocation(loaded[arm], arm));
    }
    assert(results.every(r => r.status === "complete"), "incomplete ABBA retained; no retry");
    const semantic = (r: any) => ({ plan: r.planSha256, resolved: r.resolvedSha256, candidates: r.candidates,
      hops: [...r.hops].sort((a, b) => a.trial - b.trial || a.hop - b.hop) });
    for (const r of results) assert.deepEqual(semantic(r), semantic(results[0]), "amount/output/prefix/plan parity failed");
    const storage = (r: any) => r.physical.filter((p: any) => p.method === "eth_getStorageAt");
    const words = (r: any) => [...new Set(storage(r).map((p: any) => p.keySha256 + ":" + p.resultSha256))].sort();
    for (const r of results) assert.deepEqual(words(r), words(results[0]), "storage keys/bytes changed");
    for (const b of results.filter(r => r.arm === "B")) for (const a of results.filter(r => r.arm === "A"))
      assert(storage(b).length < storage(a).length, "no actual storage call reduction");
  } catch (e) { error = clean(e); process.exitCode = 1; }
  finally {
    try {
      assert.equal(escapeAttempts, 0, "network/process escape attempt invalidates result");
      for (const item of artifacts) assert.equal(hash(item.path), item.sha256, "input drift");
      assert.deepEqual(tree(roots.A), before.A); assert.deepEqual(tree(roots.B), before.B);
      assert.equal(hash(fileURLToPath(import.meta.url)), declaration.runnerSha256);
    } catch (e) { error = clean(e); process.exitCode = 1; }
    fs.writeFileSync(path.join(out, "results.json"), json({ status: error ? "failed" : "passed", error, escapeAttempts, results }), { flag: "wx" });
  }
}
main().catch(e => { console.error(clean(e)); process.exitCode = 1; });
