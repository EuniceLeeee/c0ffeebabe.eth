// Deny sockets/child processes during imports, validation and all offline modes.
// No provider credential is even read until the declaration is durably written.
function offlineFence() {
  const connect = net.Socket.prototype.connect;
  const saved = new Map<string, any>();
  const deny = () => { throw new Error("offline fence: network/process operation forbidden"); };
  net.Socket.prototype.connect = deny as any;
  for (const key of ["spawn", "spawnSync", "exec", "execSync", "execFile", "execFileSync", "fork"]) {
    saved.set(key, (childProcess as any)[key]); (childProcess as any)[key] = deny;
  }
  syncBuiltinESMExports();
  return () => {
    net.Socket.prototype.connect = connect;
    for (const [key, value] of saved) (childProcess as any)[key] = value;
    syncBuiltinESMExports();
  };
}

function args() {
  const result: Record<string, string | boolean> = {};
  const flags = new Set(["--run", "--fixture-check", "--self-test"]);
  const values = new Set(["--spec", "--baseline-runtime", "--candidate-runtime", "--binary-a", "--binary-b", "--experiment", "--out"]);
  const argv = process.argv.slice(2);
  for (let i = 0; i < argv.length; i++) {
    const key = argv[i]!;
    assert(!(key in result), `duplicate option ${key}`);
    if (flags.has(key)) result[key] = true;
    else { assert(values.has(key), `unsupported option ${key}`); const value = argv[++i];
      assert(value && !value.startsWith("--"), `missing value ${key}`); result[key] = value; }
  }
  assert(!(result["--run"] && (result["--fixture-check"] || result["--self-test"])), "offline and run modes are exclusive");
  return result;
}

function sourceTree(root: string) {
  assert(path.isAbsolute(root), "runtime must be absolute");
  assert.equal(fs.realpathSync(root), root, "runtime root must be a real directory");
  assert.notEqual(root, repo, "working checkout is not a frozen runtime");
  const base = path.join(root, "listener/src");
  assert.equal(fs.realpathSync(base), base, "source root cannot be symlinked");
  const files: Record<string, string> = {};
  const walk = (relative: string) => {
    for (const entry of fs.readdirSync(path.join(base, relative), { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
      const name = path.posix.join(relative, entry.name);
      if (entry.isDirectory()) walk(name);
      else { assert(entry.isFile(), `non-regular source entry ${name}`); files[name] = sha(path.join(base, name)); }
    }
  };
  walk("");
  const runtimeAssets: Record<string, string> = {};
  const assets = (relative: string) => {
    const file = path.join(root, relative), stat = fs.lstatSync(file);
    if (stat.isDirectory()) for (const name of fs.readdirSync(file).sort()) assets(path.posix.join(relative, name));
    else { assert(stat.isFile(), `non-regular runtime asset ${relative}`); runtimeAssets[relative] = sha(file); }
  };
  assets("listener/searcher"); assets("listener/tsconfig.json"); assets("listener/tsconfig.live.json");
  const require = createRequire(path.join(root, "listener/package.json"));
  return { root, files, runtimeAssets, sha256: shaBytes(json(files)),
    packageSha256: sha(path.join(root, "listener/package.json")),
    dependencyRoot: fs.realpathSync(path.join(root, "listener/node_modules")),
    ethersEntry: require.resolve("ethers"), artifactSha256: sha(path.join(root, "out/BotVM.sol/BotVM.json")) };
}
function compareTree(actual: ReturnType<typeof sourceTree>, basis: ReturnType<typeof sourceTree>, policy: "A" | "B") {
  assert.deepEqual(Object.keys(actual.files), Object.keys(basis.files), "runtime file sets differ");
  const deltas = policy === "B" ? spec.runtimeDeltaSha256B ?? {} : {};
  if (Object.keys(deltas).length) assert.deepEqual(Object.keys(deltas).sort(), [
    "searcher/revm-sim-client.ts", "searcher/revm-strict-source-owner.ts", "searcher/test/revm-sim-client.ts",
  ], "only declared strict-pair transport files may differ");
  for (const key of Object.keys(basis.files)) assert.equal(actual.files[key], key === amountFile
    ? spec.amountPropagation[policy] : deltas[key] ?? basis.files[key], `unexpected runtime difference: ${key}`);
  assert.equal(actual.packageSha256, basis.packageSha256, "runtime package differs");
  assert.deepEqual(actual.runtimeAssets, basis.runtimeAssets, "runtime assets differ");
  assert.equal(actual.dependencyRoot, basis.dependencyRoot, "require identical shared installed dependencies");
  assert.equal(actual.artifactSha256, spec.artifactSha256, "executor artifact differs");
}

async function offlineRegistrationRegression(root: string, m: any, f: any) {
  // Compiler/registration regression, NOT a historical quote or latency trial.
  // Use existing Ready directions, real issuers/Family fragments/compiler, and
  // synthetic ABI responses only at the transport boundary. No custom adapters.
  const { ethers } = createRequire(path.join(root, "listener/package.json"))("ethers");
  const { UNIV4_QUOTER_INTERFACE: quoter } = await import(pathToFileURL(path.join(root,
    "listener/src/searcher/venues/swaps/univ4-abi.ts")).href);
  const token = new ethers.Interface(["function balanceOf(address) view returns (uint256)",
    "function getEthBalance(address) view returns (uint256)"]);
  const subscript = new ethers.Interface(["function execSubscript(bytes)"]);
  const { root: runtimeRoot, cases } = rehydrate(m, f);
  const saved = cases.find((item: any) => item.rank === 8);
  const edges = [saved.edges[0], saved.edges.at(-1)];
  assert.equal(edges[0].instanceKey, edges[1].instanceKey);
  let sourceReads = 0;
  const compiled: string[] = [];
  const respond = (data: string) => {
    if (data.startsWith(quoter.getFunction("quoteExactInputSingle").selector)) {
      quoter.decodeFunctionData("quoteExactInputSingle", data);
      return quoter.encodeFunctionResult("quoteExactInputSingle", [1000n, 123n]);
    }
    const call = token.parseTransaction({ data }); assert(call, "unexpected offline balance request");
    return token.encodeFunctionResult(call.name, [1n << 128n]);
  };
  const runtime = m.central.createStrictCentralAdapterRuntime({
    executor: f.input.executor, transactionOrigin: f.input.owner,
    exactQuoteCache: m.cache.createAdapterFamilyExactQuoteCache(),
    generationFence: { assertCurrent(generation: number, source: any) {
      assert.equal(generation, spec.source.generation); assert.deepEqual(source, spec.source);
    } },
    provider: {
      async call(tx: any) { sourceReads++; return respond(tx.data); },
      async getCode() { throw new Error("unexpected offline code read"); },
      async getStorage() { throw new Error("unexpected offline storage read"); },
    },
    simulator: {
      async simulate() { throw new Error("offline regression must use prefix transport"); },
      async simulatePrefix(input: any) {
        assert.deepEqual(input.source, spec.source);
        assert.equal(input.prefix.executor, f.input.executor.toLowerCase());
        assert.equal(input.prefix.inputToken, edges[0].tokenIn.toLowerCase());
        assert.equal(input.prefix.inputAmount, BigInt(saved.P));
        const bytes = ethers.getBytes(subscript.decodeFunctionData("execSubscript", input.prefix.calldata)[0]);
        assert(bytes.length > 38, "compiled prefix must contain real action bytes");
        assert.equal(bytes[0], 0x09, "actual-amount-flow opcode");
        assert.equal(bytes[36], 0, "exact prefix retains zero tolerance");
        assert.equal(bytes[37], 1, "one authenticated earlier quote is compiled");
        assert.equal(input.request.kind, "eth-call");
        compiled.push(input.prefix.calldata);
        return { data: respond(input.request.data), completion: "returned" };
      },
    },
  });
  const session = await runtimeRoot.createSession({ source: spec.source, runtime, kind: "exact", fundingAssets: [],
    requiredEdgeIds: new Set(edges.map((edge: any) => edge.canonicalEdgeId)) });
  const quote = (edge: any, amountIn: bigint, priorQuotes: any[]) => session.issueExact({
    edge: session.edges.find((item: any) => item.canonicalEdgeId === edge.canonicalEdgeId),
    amountIn, priorQuotes, executor: f.input.executor, runtimeEvidence: [],
  });
  const first = await quote(edges[0], BigInt(saved.P), []);
  assert.equal(sourceReads, 2); assert.equal(first.amountOut, 1000n);
  const second = await quote(edges[1], first.amountOut, [first]);
  assert.equal(second.amountOut, 1000n);
  assert.equal(sourceReads, 2, "nonempty-prefix reads cannot silently use source state");
  assert.equal(compiled.length, 2, "quote and payout-balance reads both receive the compiled prefix");
  assert.equal(compiled[0], compiled[1]);
  const registered = m.registry.listAll();
  const expected = m.composition.PRODUCTION_STRICT_SHADOW_ACTION_ADAPTERS;
  assert.deepEqual(registered.map((a: any) => a.id).sort(), expected.map((a: any) => a.id).sort());
  for (const action of expected) assert.equal(m.registry.get(action.id), action, "wrong runtime's adapter registry");
  const wrappers = ["actual-amount-flow", "actual-amount-step", "actual-amount-case"];
  for (const id of wrappers) assert.equal(m.registry.get(id).isWrapper, true);
  m.registry.assertDescriptorCoverage();
  return { runtime: root, registeredActionIds: registered.map((a: any) => a.id).sort(), wrappers,
    sourceReads, prefixReads: compiled.length, compiledPrefixSha256: shaBytes(compiled[0]!),
    compiledPrefixBytes: (compiled[0]!.length - 2) / 2,
    scope: "offline registration/compilation only; synthetic transport responses; no RPC or EVM execution" };
}

// Full saved publication is rehydrated. Required edges only scope normal session preparation.
function rehydrate(m: any, f: any) {
  const envelope = f.ready;
  assert(!envelope.inProgressRun, "Ready has unfinished rebuild");
  assert.equal(m.checkpoint.envelopeFingerprint(envelope), envelope.checkpointFingerprint, "checkpoint fingerprint mismatch");
  const { ready, graph } = m.ready.resolveStrictReadyRuntime(envelope.readyGeneration);
  assert.deepEqual(ready.cutoff, f.input.topologySource);
  const catalog = m.composition.PRODUCTION_STRICT_SHADOW_FAMILY_CAPABILITY_CATALOG;
  // Rehydration uses no transport: deliberately do not give the wiring the real credential.
  const wiring = m.rebuild.createRebuildWiring({ rpcUrl: "<redacted-endpoint>",
    executionIdentity: { executor: f.input.executor, transactionOrigin: f.input.owner } });
  const instances: any[] = [], funding: any[] = [];
  const memos = m.checkpoint.activeReadyMemos(envelope);
  for (const memo of memos) {
    assert(wiring.isReadyMemoDefinitionCurrent(memo), `stale Ready definition: ${memo.familyId}`);
    const id = m.identifiers.familyId(memo.familyId), family = catalog.forStrictFamily(id);
    const item = wiring.rehydrateVerifiedInstance({ memo, cutoff: ready.cutoff });
    if (family.plugin.manifest.domain === "funding") funding.push({ familyId: id, asset: item.asset });
    else instances.push(item);
  }
  const root = new m.session.StrictProductionRuntimeRoot({ catalog, readySource: ready.cutoff,
    readyGraph: graph, readyInstances: instances, readyFundingAssets: funding });
  const cases = spec.cases.map((item: any) => {
    const original = f.enumeration.opportunities[item.rank - 1];
    const edges = original.seedEdges.map((saved: any) => {
      const edge = graph.find((entry: any) => entry.canonicalEdgeId === saved.canonicalEdgeId);
      assert(edge, `canonical edge missing at rank ${item.rank}`);
      for (const key of ["adapterId", "instanceKey", "executionVariantKey"]) assert.equal(edge[key], saved[key]);
      for (const key of ["target", "tokenIn", "tokenOut"]) assert.equal(edge[key].toLowerCase(), saved[key].toLowerCase());
      return edge;
    });
    assert.equal(new Set(edges.map(m.identity.edgeInstanceKey)).size < edges.length, item.repeated);
    const refs = new m.amount.BlockScanAmountReference().prepare({ pricing: f.prices.runtime.pricing, opportunities: [original] });
    const prepared = m.refinement.prepareBlockScanCandidatesWithoutExactRefinement([original], 1,
      { probeAmountsByOpportunity: refs, admissionSpreadBps: 0 });
    assert.equal(prepared.opportunities.length, 1, "saved amount reference rejected");
    const opportunity = prepared.opportunities[0]; assert.equal(opportunity.searchSeed.searchCenter, BigInt(item.P));
    assert.equal(opportunity.searchSeed.maxInput, original.searchSeed.maxInput, "cap changed");
    return { ...item, edges, opportunity };
  });
  return { root, cases, receipt: { instances: instances.length, funding: funding.length, graphEdges: graph.length,
    cutoff: ready.cutoff, publicationSetHash: ready.publicationSetHash, graphHash: ready.graphHash,
    activations: m.composition.PRODUCTION_FAMILY_ACTIVATIONS,
    cases: cases.map((c: any) => ({ rank: c.rank, P: c.P, cap: c.opportunity.searchSeed.maxInput,
      repeated: c.repeated, edges: c.edges.map((e: any) => ({ canonicalEdgeId: e.canonicalEdgeId, instance: m.identity.edgeInstanceKey(e) })) })) } };
}

const hopContext = new AsyncLocalStorage<number>();
function expectedPrefix(policy: string, repeated: boolean, hop: number) {

function watch<T>(pending: Set<Promise<any>>, task: Promise<T>): Promise<T> {
  pending.add(task); void task.then(() => pending.delete(task), () => pending.delete(task)); return task;
}

async function invocation(m: any, f: any, item: any, experiment: any, code: any, globalAbort: AbortController) {

      allowSingleCallFallback: false, onSourceUnavailable: (error: any) => controller.abort(error),
      onRpcThrottle: (error: any) => controller.abort(error) });
    // Preserve the real batching/memo backend; fail the scope before central's bounded retry.
    const safeBackend = new Proxy(backend, { get(target, key) {
      const value = Reflect.get(target, key, target);
      if (key === "call" || key === "callCached") return (...input: any[]) => {
        // A cache miss is synchronously undefined, NOT a fulfilled Promise of undefined.

      identity: { source: spec.source, chainId: 1, rpcUrl: secret, stateRoot: spec.stateRoot }, control,
      executionGasLimit: 0x1000000, executorRuntimeCode: code, onFatal: fatal,
      createClient: ({ onFatal }: any) => {

        // inferred join from request ordering. This binds Rust phase records.
        class ObservedClient extends m.client.RevmSimClient {
          spawnDaemon(command: string, argv: string[], detached: boolean) {
            const proc = super.spawnDaemon(command, argv, detached);
            r.revmProcesses.push({ pid: proc.pid ?? null, atMs: performance.now() - started });
            return proc;
          }
        }
        const client = new ObservedClient({ executablePath: item.binary, timeoutMs: 30_000, onFatal });
        return { get isTerminal() { return client.isTerminal; }, closeAndDrain: (reason: any) => client.closeAndDrain(reason),
          strictSimulate: (request: any, work: any) => {

              requestKind: request.kind ?? null, target: request.to ?? null };
            r.revmRequests.push(metric); const start = performance.now();

              metric.ok = value.ok; metric.epoch = value.epoch; metric.requestId = value.requestId;
              metric.daemonLatencyMs = value.latencyMs; metric.cacheStats = value.cacheStats;
              metric.seedStats = value.seedStats; return value;

          } };
      },
    });
    const underlying = simulation.transport;
    const transport = { ...underlying, simulatePrefix: (input: any) => {

        kind: input.request.kind, atMs: performance.now() - started };

        .finally(() => { metric.wallMs = performance.now() - start; }));
    } };

      simulator: transport, exactQuoteCache: m.cache.createAdapterFamilyExactQuoteCache(),

      verifiedActors: m.actors.PRODUCTION_STRICT_VERIFIED_ACTORS, transportScheduler: shared,
      generationFence: { assertCurrent(generation: number, source: any) {
        assert.equal(generation, spec.source.generation); assert.deepEqual(source, spec.source); guard();
      } } });
    const actualSession = await root.createSession({ source: spec.source, runtime, kind: "exact", control,
      fundingAssets: [selected.opportunity.flashToken], requiredEdgeIds: new Set(selected.edges.map((e: any) => e.canonicalEdgeId)) });

    const session = new Proxy(actualSession, { get(target, key) {
      if (key === "issueExact") return (input: any) => {
        guard(); const hop = indices.get(input.edge.canonicalEdgeId) as number; assert.notEqual(hop, undefined);
        const metric: any = { id: r.hops.length + 1, hop, canonicalEdgeId: input.edge.canonicalEdgeId,
          amountIn: input.amountIn, prefixLength: input.priorQuotes === undefined ? null : input.priorQuotes.length,
          atMs: performance.now() - started, status: "running" };
        r.hops.push(metric);
        assert.equal(metric.prefixLength, expectedPrefix(item.policy, selected.repeated, hop), "propagation prefix policy mismatch");
        const start = performance.now();
        return hopContext.run(metric.id, () => watch(pending, target.issueExact(input).then((value: any) => {
          metric.amountOut = value.amountOut; metric.status = "quoted"; return value;
        }, (error: any) => { metric.status = "failed"; metric.error = clean(error); throw error; })
          .finally(() => { metric.wallMs = performance.now() - start; })));
      };
      const value = Reflect.get(target, key, target); return typeof value === "function" ? value.bind(target) : value;
    } });
    const opportunity = { ...selected.opportunity, seedEdges: selected.edges.map((edge: any) => {
      const found = session.edges.find((e: any) => e.canonicalEdgeId === edge.canonicalEdgeId); assert(found); return found;
    }) };

    planner.setFlashLiquidity(session.fundingLiquidityView());

      [m.templates.FLASH_LEND_SWAP_REPAY, m.templates.FLASH_SWAP_REPAY], { deadlineAtMs });
    r.planCount = plans.length; assert(plans.length, "no plan for saved candidate"); r.planIndex = 0;
    assert.deepEqual(plans[0].tokenPath.edges.map((e: any) => e.canonicalEdgeId),
      opportunity.seedEdges.map((e: any) => e.canonicalEdgeId), "Planner changed saved route/order");
    r.sessionCreationTiming = actualSession.creationTiming;
    r.planSha256 = shaBytes(json(plans[0]));
    guard(); r.preparationMs = performance.now() - started; solverStart = performance.now();
    r.stage = experiment.phase; r.solverEntered = experiment.phase === "solver";
    if (experiment.phase === "p-only") {
      const value = await m.propagation.propagateAmountsWithRawOutputs({ edges: opportunity.seedEdges }, BigInt(selected.P), safeBackend,
        { executor: f.input.executor, strictSession: session, runtimeEvidence: [], adapterWorkControl: control,
          toleranceRawUnits: 0n, shouldStop: () => { guard(); return false; } });
      r.pTrial = { amounts: value.amounts, rawOutputs: value.rawOutputs, profit: value.amounts.at(-1) - BigInt(selected.P) };
    } else {
      await new m.solver.AnvilSolver().solve(plans[0], safeBackend, { executor: f.input.executor,
        simulate: async () => { throw new Error("final simulation forbidden in quote-only diagnostic"); } },
      { ...control, deferPhase2Sim: true, finalSimTopN: 3, gridHalfWidth: 2, blockScanAmountGrid: "multiples", gssMaxTries: 8,
        quoteProfitFloorBps: 0n, quoteToleranceRawUnits: 0n, strictSession: session, runtimeEvidence: [], timing: r.timing,
        onDeferredCandidates: (values: any[]) => { r.candidates = values.map(v => ({ amount: v.flashAmount, profit: v.netProfit, profitToken: v.profitToken })); } });
    }
    r[experiment.phase === "solver" ? "solverMs" : "propagationMs"] = performance.now() - solverStart; solverStart = undefined;
    guard(); r.status = "returned";
  } catch (error) {
    r.status = controller.signal.aborted ? "cancelled" : "failed"; r.error = clean(error);
    // Semantic P rejection remains a failed economic search, not a completed full search.
  } finally {
    if (solverStart !== undefined) r[experiment.phase === "solver" ? "solverMs" : "propagationMs"] = performance.now() - solverStart;
    r.preparationMs ??= performance.now() - started;
    // Recheck the source even after a semantic P/quote rejection, if budget remains.
    if (canonical && r.canonicalBefore && !controller.signal.aborted) {
      const checkStart = performance.now();
      try { guard(); await canonical(); r.canonicalAfter = true; }
      catch (error) { r.canonicalAfter = false; r.canonicalError = clean(error);
        r.status = controller.signal.aborted ? "cancelled" : "failed"; }
      r.canonicalAfterMs = performance.now() - checkStart;
    }
    r.measuredMs = performance.now() - started; r.aborted = controller.signal.aborted;
    clearTimeout(timer); globalAbort.signal.removeEventListener("abort", abort);
    const drainStarted = performance.now(); controller.abort(new Error("invocation settled"));
    const drains = await Promise.allSettled([backend?.closeAndDrain(), simulation?.closeAndDrain()]);
    while (pending.size) await Promise.allSettled([...pending]);
    r.drainMs = performance.now() - drainStarted; r.totalWithDrainMs = performance.now() - started;
    r.backend = backend?.stats(); r.drainErrors = drains.filter(d => d.status === "rejected").map((d: any) => clean(d.reason));
    if (r.drainErrors.length) { r.status = "drain-failed"; globalAbort.abort(new Error("physical drain failed")); }
    Object.assign(console, originalConsole);
  }
  return r;
}
