import fs from 'node:fs';

const here = path.dirname(fileURLToPath(import.meta.url));
const sha = (s) => createHash('sha256').update(s).digest('hex');
const q = (values, p) => {
  const sorted = [...values].sort((a,b) => a-b);
  return sorted.length ? sorted[Math.ceil(sorted.length*p)-1] : null;
};
function canonicalQuotes(r) {
  const first = r.hops.filter(h => h.hop === 0);
  const last = r.hops.filter(h => h.hop === 1);
  assert.equal(first.length, 12); assert.equal(last.length, 12);
  assert.equal(new Set(first.map(h => h.amountIn)).size, 12);
  assert.equal(new Set(first.map(h => h.amountOut)).size, 12);
  assert.equal(new Set(last.map(h => h.amountIn)).size, 12);
  assert(r.hops.every(h => h.status === 'quoted' && h.prefixLength === null));
  // First-hop issuance preserves the production grid/GSS selection order.
  // Independent second hops can finish in another order; match their input.
  return first.map(h => {
    const next = last.find(n => n.amountIn === h.amountOut);
    assert(next, 'missing second hop for first-hop output');
    return { edges: [h.canonicalEdgeId,next.canonicalEdgeId],
      amounts: [h.amountIn,h.amountOut,next.amountOut] };
  });
}
function transport(r) {
  const dispatch = (r.transportEvents ?? []).filter(e => e.kind === 'dispatch');
  const settled = (r.transportEvents ?? []).filter(e => e.kind === 'settled');
  const calls = dispatch.flatMap(e => e.calls.map(c => JSON.stringify({
    method:e.method,source:e.sourceBlockHash,to:c.target,from:c.from,data:c.calldataSha256,
  }))).sort();
  return { calls, batches: dispatch.length, maxItems: Math.max(0,...dispatch.map(e=>e.calls.length)),
    responses: settled.map(e=>({items:e.items,wallMs:e.wallMs,status:e.status,statusCode:e.statusCode,
      headerWaitMs:e.http?.responseHeadersAtMs == null ? null : e.http.responseHeadersAtMs-e.startedAtMs})) };
}

    solverP95:q(rr.map(r=>r.solverMs).filter(v=>typeof v==='number'),.95),
    physicalEnvelopes:rr.map(r=>r.batches),logicalCalls:rr.map(r=>r.logicalCalls)};
};
