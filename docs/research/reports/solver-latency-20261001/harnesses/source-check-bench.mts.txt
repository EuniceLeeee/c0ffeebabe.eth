// Fixed-input transport microbenchmark, NOT a route/Family acceptance replay.
import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {resolve,dirname} from 'node:path';
import {fileURLToPath} from 'node:url';
import {RevmSimClient} from '../../listener/src/searcher/revm-sim-client.js';
const dir=dirname(fileURLToPath(import.meta.url)),root=resolve(dir,'../..');
const hash=(b:string|Buffer)=>createHash('sha256').update(b).digest('hex');
const bin={A:resolve(root,'logs/live-solver-unique-pools-250.YgUfBJ/runtime/revm-sim'),B:resolve(root,'logs/live-solver-batched-prefix-250.I49DJ4/runtime/revm-sim')};
assert.equal(hash(readFileSync(bin.A)),'461357a18302d74b78a79527ef360c5e3ddfc2d5d58c78981f4f487467f43a36');
assert.equal(hash(readFileSync(bin.B)),'5f9c6dba64508decdeff3f5414453f804a9bed48bc947acd2444dd39734e397c');
const preflight=JSON.parse(readFileSync(resolve(root,'logs/live-solver-batched-prefix-250.I49DJ4/executor-code-override-preflight.json'),'utf8'));
let rpc='';
for(const line of readFileSync('<user>/src/MEV/.env','utf8').split(/\r?\n/)){
  const m=line.trim().match(/^(?:export\s+)?MAINNET_RPC_URL=(.*)$/);
  if(m)rpc=m[1]!.trim().replace(/^(['"])(.*)\1$/,'$2');
}
assert.equal(hash(rpc),'860a9610cd135f044f125b2364ad22718483029de2022075cd8619cbd8ba6b4a');
const request={blockNumber:preflight.source.number,sourcePin:{chainId:1,blockHash:preflight.source.hash},rpcUrl:rpc,
  from:preflight.owner,to:'0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2',data:'0x313ce567',gasLimit:100000};
const order=Array.from({length:20},(_,i)=>i%2===0?['A','B'] as const:['B','A'] as const);
const record=(name:string,v:unknown)=>writeFileSync(resolve(dir,name),JSON.stringify(v,null,2)+'\n',{flag:'wx',mode:0o600});
record('source-check-bench-declaration.json',{source:preflight.source,scope:'read-only WETH decimals through production strict client; source-check cost only, not full prefix or Solver speed',pairs:20,order,warmupsPerLane:1,
  controls:'same full source pin and unchanged request; no deals, overrides, funding, signing or broadcast; return data/gas/attestation must match',
  binaries:{A:hash(readFileSync(bin.A)),B:hash(readFileSync(bin.B))},rpcIdentity:hash(rpc),wallLimitMs:180000,at:new Date().toISOString()});
const clients={A:new RevmSimClient({executablePath:bin.A,timeoutMs:12000}),B:new RevmSimClient({executablePath:bin.B,timeoutMs:12000})};
const control=new AbortController(),timer=setTimeout(()=>control.abort(),180000);
const rows:any[]=[],warmups:any[]=[];let expected:unknown,failed=false;
async function run(lane:'A'|'B',pair:number){
  const start=performance.now();
  try{
    const r=await clients[lane].strictSimulate(request,{signal:control.signal,deadlineAtMs:Date.now()+12000});
    assert(r.ok&&r.success&&r.output===`0x${'0'.repeat(62)}12`);
    const facts={output:r.output,gasUsed:r.gasUsed,strict:r.strict,attestation:r.sourceAttestation};
    if(expected===undefined)expected=facts;else assert.deepEqual(facts,expected);
    return {lane,pair,ms:performance.now()-start,status:'pass'};
  }catch(error){failed=true;return {lane,pair,ms:performance.now()-start,status:'fail',category:error instanceof Error?error.name:'unknown'};}
}
try{
  for(const lane of ['A','B'] as const){warmups.push(await run(lane,-1));if(failed)break;}
  if(!failed)for(let i=0;i<order.length;i++){
    for(const lane of order[i]!){rows.push(await run(lane,i));if(failed)break;}
    console.log(JSON.stringify({pair:i+1,status:failed?'failed':'pass',samples:rows.length}));if(failed)break;
  }
}finally{
  control.abort();clearTimeout(timer);await Promise.allSettled(Object.values(clients).map(c=>c.closeAndDrain()));
  const stats=(lane:string)=>{const a=rows.filter(r=>r.lane===lane&&r.status==='pass').map(r=>r.ms).sort((a,b)=>a-b);return{n:a.length,p50:a[Math.ceil(a.length*.5)-1]??null,p95:a[Math.ceil(a.length*.95)-1]??null};};
  const result={at:new Date().toISOString(),failed,warmups,rows,A:stats('A'),B:stats('B'),facts:expected,scope:'fixed read-only request; excludes startup warmups; all attempted trials retained; not whole Solver/prefix-route latency'};
  record('source-check-bench-result.json',result);console.log(JSON.stringify(result));if(failed)process.exitCode=1;
}
