// One write-once censored-phase diagnostic. No signing or submission.
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { createRequire } from 'node:module';
import { createHash } from 'node:crypto';
import assert from 'node:assert/strict';
import { runOwned } from './launch-cold-source.mjs';

const here = path.dirname(fileURLToPath(import.meta.url));
const repo = path.resolve(here, '../..');
async function main() {
  const specPath = path.join(here, 'prefix-progress-timing.json');
  const spec = JSON.parse(fs.readFileSync(specPath, 'utf8'));
  assert.equal(spec.predeclaredAssessment.attempts, 1);
  assert.equal(spec.budgetMs, 30000);
  assert.deepEqual(spec.experiments['prefix-phases'].order, ['B']);
  const binary = spec.diagnosticBinary;
  assert.equal(typeof binary, 'string');
  assert.equal(path.dirname(path.dirname(binary)), here);
  assert.match(path.basename(path.dirname(binary)), /^prefix-progress-binary\.[A-Za-z0-9]+$/);
  assert.equal(path.basename(binary), 'revm-sim');
  assert.equal(createHash('sha256').update(fs.readFileSync(binary)).digest('hex'), spec.binarySha256.current);
  const out = path.join(here, 'same-input-prefix-progress-1');
  assert(!fs.existsSync(out), 'never overwrite an attempted diagnostic');
  const logPath = path.join(here, 'same-input-prefix-progress-1.log');
  const receiptPath = path.join(here, 'prefix-progress-launch-receipt.json');
  assert(!fs.existsSync(receiptPath));
  const fd = fs.openSync(logPath, 'wx', 0o600);
  let rpc = '';
  for (const line of fs.readFileSync('<user>/src/MEV/.env', 'utf8').split(/\r?\n/)) {
    const match = line.trim().match(/^(?:export\s+)?MAINNET_RPC_URL=(.*)$/);
    if (match) rpc = match[1].trim().replace(/^(['"])(.*)\1$/, '$2');
  }
  assert.equal(createHash('sha256').update(rpc).digest('hex'),
    '860a9610cd135f044f125b2364ad22718483029de2022075cd8619cbd8ba6b4a');
  const require = createRequire(path.join(repo, 'listener/package.json'));
  const result = await runOwned(['--import', pathToFileURL(require.resolve('tsx')).href,
    path.join(here, 'same-input-pair-timing.mts'), '--spec', specPath, '--experiment', 'prefix-phases',
    '--candidate-runtime', path.join(here, 'prefix-pair-runtime-complete'),
    '--binary-a', binary, '--binary-b', binary, '--run', '--out', out], {
    cwd: repo, stdio: ['ignore', fd, fd],
    env: { PATH: process.env.PATH, MAINNET_RPC_URL: rpc,
      SEARCHER_STATE_LATENCY_DIAGNOSTICS: '1', SEARCHER_DRY_RUN: '1' },
  }, 45000, 55000);
  fs.closeSync(fd);
  const interrupted = result.forcedStop || result.softStop || result.residualGroupStopped;
  fs.writeFileSync(receiptPath, JSON.stringify({ ...result, out, logPath,
    diagnosticCompleted: !interrupted && result.code === 0,
    claim: 'One selected P-only phase diagnostic. A normal exit can retain a cancelled route and never proves a Solver speedup.' }, null, 2),
    { flag: 'wx', mode: 0o600 });
  console.log(JSON.stringify({ ...result, out, logPath }));
  process.exitCode = interrupted ? 1 : result.code ?? 1;
}
if (process.argv[1] && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  main().catch(() => { console.error('prefix progress diagnostic launcher failed'); process.exitCode = 1; });
}
